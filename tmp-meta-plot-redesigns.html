simulate_smd_meta <- function(k = 12, n_min = 30, n_max = 100, m1 = 0.4, m2 = 0, sd1 = 1, sd2 = 1, method = "REML") {
  dat <- data.frame(yi = numeric(0), vi = numeric(0), n = integer(0), study = character(0))
  for (i in seq_len(k)) {
    n_i <- sample(n_min:n_max, 1)
    x <- rnorm(n_i, mean = m1, sd = sd1)
    y <- rnorm(n_i, mean = m2, sd = sd2)
    es <- metafor::escalc(
      n1i = n_i, n2i = n_i,
      m1i = mean(x), m2i = mean(y),
      sd1i = sd(x), sd2i = sd(y),
      measure = "SMD"
    )
    dat[i, "yi"] <- es$yi
    dat[i, "vi"] <- es$vi
    dat[i, "n"] <- n_i
    dat[i, "study"] <- paste("Study", i)
  }
  res <- metafor::rma(yi, vi, data = dat, method = method)
  list(dat = dat, res = res)
}

simulate_or_meta <- function(k = 12, n_min = 30, n_max = 80, pr1 = 0.70, pr2 = 0.50) {
  ai <- numeric(k)
  bi <- numeric(k)
  ci <- numeric(k)
  di <- numeric(k)
  n_each <- integer(k)

  for (i in seq_len(k)) {
    n_i <- sample(n_min:n_max, 1)
    x <- rbinom(n_i, 1, pr1)
    y <- rbinom(n_i, 1, pr2)
    ai[i] <- sum(x == 1)
    bi[i] <- sum(x == 0)
    ci[i] <- sum(y == 1)
    di[i] <- sum(y == 0)
    n_each[i] <- n_i
  }

  dat <- data.frame(ai = ai, bi = bi, ci = ci, di = di, n = n_each, study = paste("Study", seq_len(k)))
  es <- metafor::escalc(measure = "OR", ai = ai, bi = bi, ci = ci, di = di, data = dat)
  dat$yi <- es$yi
  dat$vi <- es$vi
  res <- metafor::rma(yi, vi, data = dat)
  list(dat = dat, res = res)
}

make_heterogeneity_example <- function(k = 30) {
  dat <- data.frame(yi = numeric(0), vi = numeric(0), subgroup = character(0), study = character(0))

  for (i in seq_len(k / 2)) {
    n_i <- sample(30:100, 1)
    x <- rnorm(n_i, 0.5, 1)
    y <- rnorm(n_i, 0.0, 1)
    es <- metafor::escalc(n1i = n_i, n2i = n_i, m1i = mean(x), m2i = mean(y), sd1i = sd(x), sd2i = sd(y), measure = "SMD")
    dat[i, 1:2] <- es
    dat[i, "subgroup"] <- "Manipulation A"
    dat[i, "study"] <- paste("Study", i)
  }

  for (i in (k / 2 + 1):k) {
    n_i <- sample(30:100, 1)
    x <- rnorm(n_i, 0.2, 1)
    y <- rnorm(n_i, 0.0, 1)
    es <- metafor::escalc(n1i = n_i, n2i = n_i, m1i = mean(x), m2i = mean(y), sd1i = sd(x), sd2i = sd(y), measure = "SMD")
    dat[i, 1:2] <- es
    dat[i, "subgroup"] <- "Manipulation B"
    dat[i, "study"] <- paste("Study", i)
  }

  dat <- dat[sample(nrow(dat)), ]
  rownames(dat) <- NULL
  res <- metafor::rma(yi, vi, data = dat)
  list(dat = dat, res = res)
}

forest_df_from_rma <- function(res, dat, level = 0.95, label_col = "study") {
  z <- qnorm(1 - (1 - level) / 2)
  out <- data.frame(
    study = dat[[label_col]],
    yi = dat$yi,
    vi = dat$vi,
    sei = sqrt(dat$vi)
  )
  out$ci_lb <- out$yi - z * out$sei
  out$ci_ub <- out$yi + z * out$sei
  out$weight <- 1 / out$vi
  out$weight_scaled <- out$weight / max(out$weight)
  out$order <- rev(seq_len(nrow(out)))

  pooled <- data.frame(
    study = "Pooled",
    yi = as.numeric(res$b[1]),
    ci_lb = as.numeric(res$ci.lb),
    ci_ub = as.numeric(res$ci.ub),
    order = 0
  )

  list(studies = out, pooled = pooled)
}

plot_journal_ggforest <- function(res, dat, xlab = "Hedges g", title = "Custom ggplot forest") {
  fd <- forest_df_from_rma(res, dat)
  p <- ggplot(fd$studies, aes(y = order, x = yi)) +
    geom_vline(xintercept = 0, linetype = "dashed", linewidth = 0.5, color = "#777777") +
    geom_segment(aes(x = ci_lb, xend = ci_ub, yend = order), linewidth = 0.55, color = "#303030") +
    geom_point(aes(size = weight_scaled), shape = 15, color = "#111111") +
    geom_segment(data = fd$pooled, aes(x = ci_lb, xend = ci_ub, y = order, yend = order), linewidth = 1.2, color = "#1B4D3E") +
    geom_point(data = fd$pooled, aes(x = yi, y = order), color = "#1B4D3E", shape = 18, size = 3.5, inherit.aes = FALSE) +
    scale_y_continuous(
      breaks = c(fd$studies$order, 0),
      labels = c(fd$studies$study, "Pooled"),
      expand = expansion(mult = c(0.02, 0.06))
    ) +
    scale_size_continuous(range = c(1.8, 5.5), guide = "none") +
    labs(x = xlab, y = NULL, title = title) +
    coord_cartesian(clip = "off") +
    theme_bw(base_size = 11) +
    theme(
      panel.grid.major.y = element_blank(),
      panel.grid.minor = element_blank(),
      plot.title = element_text(face = "bold"),
      axis.text.y = element_text(size = 9)
    )

  p
}

Goal

This temporary file does two things:

  1. Reproduces the current style of forest plots used in the book (“old”).
  2. Shows redesign options that are clearer and journal-oriented with minimal dependencies.

Package policy used here:

  • No new required dependency for redesigns (uses metafor + ggplot2, already used in the book).
  • One optional extra package explored: meta.

Recommendation Summary

Based on readability, flexibility, and dependency constraints:

  1. If you want minimal change and no new dependencies: use Option 1 (cleaned metafor::forest).
  2. If you want the highest visual quality and consistency with your ggplot figures: use Option 2 (custom ggplot2 forest template).
  3. If you are open to one extra dependency for forest-plot ergonomics: try Option 3 (meta package) and compare.
